import math

def compute_confidence_score(raw_value: float, weight: float = 1.0, normalize: bool = False) -> float:
    """
    Compute a confidence score from a raw value.
    
    Args:
        raw_value: The input value to convert to confidence
        weight: Multiplier applied to raw_value (default: 1.0)
        normalize: If True, constrains output to [0, 1] range using sigmoid transformation
    
    Returns:
        Confidence score as float
    """
    # Apply weight to raw value
    weighted = raw_value * weight
    
    if normalize:
        # Use tanh-based sigmoid for numerical stability
        # Maps (-inf, +inf) -> (0, 1)
        return 0.5 * (math.tanh(weighted) + 1)
    else:
        return weighted

**Key design decisions:**

- Weight parameter allows scaling before transformation

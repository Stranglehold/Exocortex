```python
def compute_confidence_score(raw_value: float, weight: float = 1.0, normalize: bool = False) -> float:
    """
    Compute a confidence score from a raw value.
    
    Args:
        raw_value: The base value to compute confidence from
        weight: A multiplier for the raw value (default: 1.0)
        normalize: If True, apply sigmoid normalization to [0, 1] range
    
    Returns:
        A float representing the confidence score
    """
    # Apply weight to raw value
    weighted_value = raw_value * weight
    
    # Normalize to [0, 1] using sigmoid if requested
    if normalize:
        return 1.0 / (1.0 + math.exp(-weighted_value))
    
    return weighted_value
```

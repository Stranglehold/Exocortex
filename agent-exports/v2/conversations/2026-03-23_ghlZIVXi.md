def compute_confidence_score(raw_value: float, weight: float = 1.0, normalize: bool = False) -> float:
    """
    Compute a confidence score from a raw value with optional weighting and normalization.
    
    Args:
        raw_value: The raw input value to convert into a confidence score.
        weight: A multiplier applied to the raw value (default is 1.0).
        normalize: If True, normalizes the result to the range [0, 1] using sigmoid transformation.
    
    Returns:
        float: The computed confidence score.
    
    Examples:
        >>> compute_confidence_score(2.5)
        2.5
        >>> compute_confidence_score(2.5, weight=2.0)
        5.0
        >>> compute_confidence_score(-1.0, normalize=True)
        0.2689414213699951
    """
    import math

    # Apply weighting to the raw value
    weighted_value = raw_value * weight

    if normalize:
        # Use sigmoid function to map any real number to (0, 1) range
        # This is suitable for confidence scores as it smoothly transitions between 0 and 1
        try:
            normalized_score = 1 / (1 + math.exp(-weighted_value))
        except OverflowError:
            # Handle extreme values gracefully
            return 1.0 if weighted_value > 0 else 0.0
        return normalized_score
    
    return weighted_value
